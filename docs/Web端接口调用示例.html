健康检查 采集源 采集任务 采集结果 JS调用示例

Collector Service - Web端接口调用示例

Base URL: http://localhost:5001/api/v1

注意: 以下示例假设前端页面运行在不同端口，需要CORS支持（已通过flask-cors配置）

2. 采集源管理

2.1 获取采集源列表

GET /api/v1/sources

Query参数:

page页码（默认1）
pageSize每页数量（默认20）
keyword关键词搜索（匹配名称或编码）
platform平台筛选（如1688、taobao）
status状态筛选（enabled/disabled）

请求示例:

// 获取1688平台的采集源，第1页，每页10条
fetch('http://localhost:5000/api/v1/sources?platform=1688&page=1&pageSize=10')
  .then(res => res.json())
  .then(console.log)

2.2 获取采集源详情

GET /api/v1/sources/detail?sourceId=xxx

2.3 创建采集源

POST /api/v1/sources/create

请求体:

{
  "sourceCode": "SRC_1688_DEFAULT",    // 必填，采集源编码
  "sourceName": "1688默认采集源",      // 必填，采集源名称
  "platform": "1688",                 // 平台
  "sourceType": "url",                 // 类型：url/shop/keyword
  "entryUrl": "https://xindeyi.1688.com",  // 入口URL
  "parserCode": "1688_offer_v1",
  "config": {                         // 配置项
    "rateLimitPerMinute": "60",
    "timeoutSeconds": "30",
    "retryTimes": "2",
    "saveImagesToOss": true
  }
}

请求示例:

const response = await fetch('http://localhost:5000/api/v1/sources/create', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    sourceCode: 'SRC_1688_DEFAULT',
    sourceName: '1688默认采集源',
    platform: '1688',
    sourceType: 'url',
    entryUrl: 'https://xindeyi.1688.com'
  })
});
const data = await response.json();
console.log(data);

3. 采集任务

3.1 创建采集任务

POST /api/v1/tasks/create

请求体:

{
  "taskType": "shop",              // 必填：single_url/batch_url/shop/keyword
  "sourceId": "src_14871529",    // 采集源ID
  "targetUrls": [],                // batch_url时使用，批量URL列表
  "options": {
    "maxItems": "200",            // 最大采集数量
    "concurrency": "3",         // 并发数
    "saveImagesToOss": true
  },
  "operatorId": "user_001",      // 操作人ID
  "idempotencyKey": "key_xxx"  // 幂等键（可选）
}

请求示例:

const response = await fetch('http://localhost:5000/api/v1/tasks/create', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    taskType: 'shop',
    sourceId: 'src_14871529',
    options: { maxItems: '200' }
  })
});
const data = await response.json();
// 返回: { taskId: "task_xxx", taskNo: "COL20260427XXXX", status: "pending" }

5. 完整JS调用示例（带错误处理）

const BASE_URL = 'http://localhost:5000/api/v1';

// 通用请求封装
async function request(url, options = {}) {
  try {
    const response = await fetch(`${BASE_URL}${url}`, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options.headers
      }
    });

    const result = await response.json();

    if (result.code !== 'OK') {
      throw new Error(result.message || '请求失败');
    }

    return result.data;
  } catch (error) {
    console.error('API请求错误:', error);
    throw error;
  }
}

// 健康检查
async function checkHealth() {
  return await request('/health');
}

// 获取采集源列表
async function getSources(params = {}) {
  const query = new URLSearchParams(params).toString();
  return await request(`/sources?${query}`);
}

// 创建采集源
async function createSource(data) {
  return await request('/sources/create', {
    method: 'POST',
    body: JSON.stringify(data)
  });
}

// 创建采集任务
async function createTask(data) {
  return await request('/tasks/create', {
    method: 'POST',
    body: JSON.stringify(data)
  });
}

// 获取任务列表
async function getTasks(params = {}) {
  const query = new URLSearchParams(params).toString();
  return await request(`/tasks?${query}`);
}

// 获取任务详情
async function getTaskDetail(taskId) {
  return await request(`/tasks/detail?taskId=${taskId}`);
}

// 取消任务
async function cancelTask(taskId) {
  return await request('/tasks/cancel', {
    method: 'POST',
    body: JSON.stringify({ taskId })
  });
}

// 获取采集结果列表
async function getItems(params = {}) {
  const query = new URLSearchParams(params).toString();
  return await request(`/items?${query}`);
}

// 使用示例
async function main() {
  // 1. 检查服务状态
  const health = await checkHealth();
  console.log('服务状态:', health.status);

  // 2. 获取采集源列表
  const sources = await getSources({ platform: '1688' });
  console.log('采集源:', sources.items);

  // 3. 创建任务
  const task = await createTask({
    taskType: 'shop',
    sourceId: 'src_14871529',
    options: { maxItems: '100' }
  });
  console.log('任务已创建:', task.taskId);

  // 4. 轮询任务状态
  const detail = await getTaskDetail(task.taskId);
  console.log('任务详情:', detail);
}

main();

6. 错误码说明

错误码说明
COLLECTOR_SOURCE_CODE_REQUIRED采集源编码不能为空
COLLECTOR_SOURCE_NAME_REQUIRED采集源名称不能为空
COLLECTOR_SOURCE_CODE_EXISTS采集源编码已存在
COLLECTOR_SOURCE_NOT_FOUND采集源不存在
COLLECTOR_SOURCE_DISABLED采集源已停用
COLLECTOR_TASK_TYPE_REQUIRED任务类型不能为空
COLLECTOR_TASK_TYPE_INVALID任务类型不正确
COLLECTOR_TASK_NOT_FOUND任务不存在
COLLECTOR_TASK_NOT_CANCELABLE当前状态不允许取消
COLLECTOR_TASK_NOT_RETRYABLE当前状态不允许重试
COLLECTOR_ITEM_ID_REQUIRED结果ID不能为空
COLLECTOR_ITEM_NOT_FOUND采集结果不存在
