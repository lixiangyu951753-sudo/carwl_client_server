## 一、客户端心跳请求改造

### 改造前（旧版）

```python
import requests

HEARTBEAT_URL = "http://127.0.0.1:5001/client/heartbeat"

def heartbeat():
    data = {
        "client_id": "client_01",
        "status": "idle",
        "client_type": "shop_price"  # 只传任务类型
    }
    resp = requests.post(HEARTBEAT_URL, json=data)
    return resp.json()
```

### 改造后（新版）

```python
import requests

HEARTBEAT_URL = "http://127.0.0.1:5001/client/heartbeat"

def heartbeat():
    data = {
        "client_id": "client_1688_price_01",
        "status": "idle",
        "client_type": "shop_price",  # 兼容旧版，可保留
        "client_capabilities": {      # 新增：平台 + 能力
            "platform": "1688",
            "capabilities": ["price"]
        }
    }
    resp = requests.post(HEARTBEAT_URL, json=data)
    return resp.json()
```

# client_1688_image.py

import requests
import time

class Client1688Image:
    def __init__(self):
        self.client_id = "client_1688_image_01"
        self.heartbeat_url = "http://127.0.0.1:5001/client/heartbeat"
        self.task_report_url = "http://127.0.0.1:5001/client/task_report"
        self.task_result_url = "http://127.0.0.1:5001/client/task_result"
        
        # 新增：声明客户端能力
        self.capabilities = {
            "platform": "1688",           # 支持的平台
            "capabilities": ["image"]     # 支持的能力（图片采集）
        }
    
    def heartbeat(self):
        """发送心跳请求"""
        data = {
            "client_id": self.client_id,
            "status": "idle",
            "client_type": "shop_image",  # 兼容旧版
            "client_capabilities": self.capabilities  # 新版权能
        }
        
        try:
            resp = requests.post(self.heartbeat_url, json=data, timeout=5)
            result = resp.json()
            
            if result.get("code") == 200:
                instruction = result.get("data", {})
                if instruction.get("instruction") == "start_crawl":
                    # 接收到任务
                    self.execute_task(instruction)
                else:
                    # 无任务，等待后继续轮询
                    time.sleep(3)
            else:
                print(f"心跳失败：{result.get('message')}")
                time.sleep(5)
        except Exception as e:
            print(f"心跳异常：{e}")
            time.sleep(5)
    
    def execute_task(self, instruction):
        """执行采集任务"""
        task_id = instruction.get("task_id")
        params = instruction.get("params", {})
        
        print(f"开始执行任务：{task_id}")
        print(f"采集能力：image（图片采集）")
        print(f"采集参数：{params}")
        
        # TODO: 实现你的图片采集逻辑
        # 注意：这是图片采集客户端，只采集图片，不采集价格
        
        # 示例伪代码：
        # shop_url = params.get("shop_url")
        # max_items = params.get("max_items")
        # 
        # products = []
        # for item in crawl_shop_images(shop_url, max_items):
        #     products.append({
        #         "url": item["url"],
        #         "title": item["title"],
        #         "images": item["images"],  # 图片 URL 列表
        #         # 价格字段可以为空或占位符
        #         "priceMin": "",
        #         "priceMax": ""
        #     })
        # 
        # # 上报结果
        # self.report_result(task_id, products)
    
    def report_progress(self, task_id, status, progress):
        """上报任务进度"""
        data = {
            "client_id": self.client_id,
            "task_id": task_id,
            "status": status,
            "progress": progress
        }
        requests.post(self.task_report_url, json=data)
    
    def report_result(self, task_id, products):
        """上报采集结果"""
        data = {
            "client_id": self.client_id,
            "task_id": task_id,
            "products": products
        }
        resp = requests.post(self.task_result_url, json=data)
        print(f"任务完成，上报 {len(products)} 个商品")
    
    def run(self):
        """启动客户端"""
        print(f"客户端启动：{self.client_id}")
        print(f"支持平台：{self.capabilities['platform']}")
        print(f"支持能力：{self.capabilities['capabilities']}")
        
        while True:
            self.heartbeat()


if __name__ == "__main__":

## 五、通用客户端改造（支持多能力）

如果你的客户端同时支持多种能力（如既采集价格又采集图片）：

```python
class ClientMultiCapability:
    def __init__(self):
        self.client_id = "client_1688_full_01"
        self.capabilities = {
            "platform": "1688",
            "capabilities": ["price", "image"]  # 支持多种能力
        }
    
    def heartbeat(self):
        data = {
            "client_id": self.client_id,
            "status": "idle",
            "client_capabilities": self.capabilities
        }
        # ... 发送请求
        
        # 这个客户端会接收到：
        # - platform=1688 且 capability=price 的任务
        # - platform=1688 且 capability=image 的任务
        # - platform 未设置或为空的任务（兼容旧版）
```

## 六、兼容性说明

### 旧版客户端（未修改）

```python
# 只传 client_type，不传 client_capabilities
data = {
    "client_id": "client_old_01",
    "client_type": "shop_price"
}
```

**行为**：
- ✅ 可以正常工作
- ✅ 会接收到 `taskType=shop_price` 的任务
- ✅ 也会接收到未设置 `platform`/`capability` 的任务
- ⚠️ 不会接收到指定了 `platform`/`capability` 的任务（除非匹配）

### 新版客户端（推荐）

```python
data = {
    "client_id": "client_new_01",
    "client_type": "shop_price",  # 可选，兼容旧版
    "client_capabilities": {
        "platform": "1688",
        "capabilities": ["price"]
    }
}
```

**行为**：
- ✅ 精确匹配平台和能力
- ✅ 优先接收指定 platform/capability 的任务
- ✅ 也会接收未设置 platform/capability 的任务（兼容旧版任务）

1. **修改心跳请求**：添加 `client_capabilities` 字段

## 八、常见问题

### Q1: 旧版客户端还能用吗？
**A**: 可以，完全兼容。旧版客户端会接收到：
- `taskType` 匹配的任务
- 未设置 `platform`/`capability` 的任务

### Q2: 如果一个客户端支持多个平台怎么办？
**A**: 可以设置多个 `capabilities`，或者部署多个客户端实例：
```python
# 方案 1：单个客户端支持多平台
capabilities = {
    "platform": "",  # 不限制平台
    "capabilities": ["price", "image"]
}

# 方案 2：多个客户端实例（推荐）
# client_1688_price_01
# client_jd_price_01
# client_pdd_price_01
```

### Q3: 如何验证任务分配是否正确？
**A**: 查看任务详情中的 `clientId` 字段：
```python
GET /api/v1/tasks/detail?taskId=task_xxx
```
响应中的 `clientId` 应该是接收任务的客户端 ID。
